=================== ./reactapp/src/components/Stopwatch.jsx ===================
import React, { Component } from 'react';

class Stopwatch extends Component {
  constructor(props) {
    super(props);
    this.state = {
      time: 0,
      isRunning: false,
      laps: []
    };
    this.interval = null;
    this.startTime = 0;
  }

  componentDidMount() {
    this.interval = null;
  }

  componentDidUpdate(prevProps, prevState) {
    if (!prevState.isRunning && this.state.isRunning) {
      this.startTime = Date.now() - this.state.time;
      this.interval = setInterval(() => {
        this.setState({
          time: Date.now() - this.startTime
        });
      }, 10);
    }

    if (prevState.isRunning && !this.state.isRunning && this.interval) {
      clearInterval(this.interval);
      this.interval = null;
    }
  }

  componentWillUnmount() {
    if (this.interval) {
      clearInterval(this.interval);
    }
  }

  formatTime = (milliseconds) => {
    const minutes = Math.floor(milliseconds / 60000);
    const seconds = Math.floor((milliseconds % 60000) / 1000);
    const centiseconds = Math.floor((milliseconds % 1000) / 10);

    return `${String(minutes).padStart(2, '0')}:${String(seconds).padStart(2, '0')}.${String(centiseconds).padStart(2, '0')}`;
  };

  handleStart = () => {
    this.setState({
      isRunning: true
    });
  };

  handleStop = () => {
    this.setState((prevState) => ({
      isRunning: false,
      laps: [...prevState.laps, prevState.time]
    }));
  };

  handleReset = () => {
    this.setState({
      time: 0,
      isRunning: false,
      laps: []
    });
  };

  handleLap = () => {
    this.setState((prevState) => ({
      laps: [...prevState.laps, prevState.time]
    }));
  };

  render() {
    const { time, isRunning, laps } = this.state;

    return (
      <div className="stopwatch">
        <h2>Stopwatch</h2>

        <div className="time">
          {this.formatTime(time)}
        </div>

        <div className="buttons">
          <button
            className="start"
            onClick={this.handleStart}
            disabled={isRunning}
          >
            Start
          </button>

          <button
            className="stop"
            onClick={this.handleStop}
            disabled={!isRunning}
          >
            Stop
          </button>

          <button
            className="reset"
            onClick={this.handleReset}
            disabled={isRunning}
          >
            Reset
          </button>
        </div>

        <button
          className="lap"
          onClick={this.handleLap}
          disabled={!isRunning}
        >
          Add Lap
        </button>

        <div className="laps">
          {laps.map((lap, index) => (
            <div className="lap-item" key={index}>
              {this.formatTime(lap)}
            </div>
          ))}
        </div>
      </div>
    );
  }
}

export default Stopwatch;

const reportWebVitals = onPerfEntry => {
  if (onPerfEntry && onPerfEntry instanceof Function) {
    import('web-vitals').then(({ getCLS, getFID, getFCP, getLCP, getTTFB }) => {
      getCLS(onPerfEntry);
      getFID(onPerfEntry);
      getFCP(onPerfEntry);
      getLCP(onPerfEntry);
      getTTFB(onPerfEntry);
    });
  }
};

import React from 'react';
import { render, fireEvent, getByAltText, getAllByText } from '@testing-library/react';
import Stopwatch from '../components/Stopwatch';

describe('Stopwatch Component', () => {
  it('renders_Stopwatch_without_crashing', () => {
    const {getByText} = render(<Stopwatch />);
    const stopwatch = getByText('Stopwatch');
    expect(stopwatch).toBeInTheDocument();
  });

  it('starts_the_stopwatch_when_Start_button_is_clicked', () => {
    const { getByText } = render(<Stopwatch />);
    const startButton = getByText('Start');
    fireEvent.click(startButton);
    expect(startButton).toBeDisabled(); 
  });

  it('stops_the_stopwatch_when_Stop_button_is_clicked', () => {
    const { getByText } = render(<Stopwatch />);
    const startButton = getByText('Start');
    fireEvent.click(startButton);
    const stopButton = getByText('Stop');
    fireEvent.click(stopButton);
    expect(stopButton).toBeDisabled(); 
  });

  it('adds_a_lap_when_Add_Lap_button_is_clicked', () => {
    const { getByText, container } = render(<Stopwatch />);
    const startButton = getByText('Start');
    fireEvent.click(startButton);
  
    const addLapButton = getByText('Add Lap');
    fireEvent.click(addLapButton);
    
  });


  it('resets_the_stopwatch_when_Reset_button_is_clicked', () => {
    const { getByText } = render(<Stopwatch />);
    const startButton = getByText('Start');
    fireEvent.click(startButton);

    const resetButton = getByText('Reset');
    fireEvent.click(resetButton);
    const timeDisplay = getByText('00:00.00');
    expect(timeDisplay).toBeInTheDocument();
  });

  it('formats_time_correctly', () => {
    const { getByText } = render(<Stopwatch />);
    const formattedTime = getByText('00:00.00');
    expect(formattedTime).toBeInTheDocument();
  });

  it('toggles_the_is_Running_state_when_start_and_stop_buttons_are_clicked', () => {
    const { getByText } = render(<Stopwatch />);
    const startButton = getByText('Start');
    fireEvent.click(startButton);
    const stopButton = getByText('Stop');
    fireEvent.click(stopButton);
    expect(startButton).not.toBeDisabled();
  });

  it('disables_Add_Lap_button_when_stopwatch_is_not_running', () => {
    const { getByText } = render(<Stopwatch />);
    const addLapButton = getByText('Add Lap');
    expect(addLapButton).toBeDisabled();
  });

  it('disables_Stop_button_when_stopwatch_is_not_running', () => {
    const { getByText } = render(<Stopwatch />);
    const stopButton = getByText('Stop');
    expect(stopButton).toBeDisabled();
  });

  
  
  it('disables_Add_Lap_button_after_stopping_the_stopwatch', () => {
    const { getByText } = render(<Stopwatch />);
    const startButton = getByText('Start');
    fireEvent.click(startButton);
  
    const stopButton = getByText('Stop');
    fireEvent.click(stopButton);
  
    const addLapButton = getByText('Add Lap');
    expect(addLapButton).toBeDisabled();
  });
  
});
